    const student=students.find(s=>s.id===studentID);
    if(!student){
        return res.status(404).json({message:"student not found in the database"});
    }
    student.name=req.body.name || student.name;
    student.course=req.body.course || student.course;

    res.json({message:"student data updated sucessfully",student });
});
app.delete('/student/:id',(req,res)=>{
    const studentID=parseInt(req.params.id);

    const deletestudent=students.splice(index,1);

const users=[];

export async function createUser(username,password){
    const existinguser=users.find(u=>u.username===username);
    if(existinguser){
        throw new error("user already exists in the data base");
        
    }
    const hashedPassword=await bcrypt.hash(password,10);
    const newuser={id:users.length+1,username,password:hashedPassword};

    users.push(newuser);
    return{id:newuser.id,username:newuser.username};
}

export async function findUserByUsername(username){
    return users.find(u=>u.username===username);
}
jwt.js
import jwt from 'jsonwebtoken';

const SECERET_KEY="my_spuer_secret_key";

export function generatorToken(user){
    return jwt.sign(
        {id:user.id,username:user.username},
        SECERET_KEY,
        {expiresIn:'1h'}
    );
}
export function authenticateToken(req,res,next){
    const authHeader=req.headers['authorization'];
    const token=authHeader && authHeader.split('')[1];

    if(!token){
        return res.status(401).json({message:'access token mesing or unauthoried '});
    }

    jwt.verify(token,SECERET_KEY,(err,user)=>{
        if(err){
            return res.status(401).json({message:'token expired or invalid '});
        }
        req.user=user;
        next();
    });
}

10.Create a class component that accept the date in the form of State and displays the date. (Passing the state to Class Component, Props vs State).

import React, { Component } from 'react';

class App extends Component {
  // 1. Initialize state inside the constructor
  constructor(props) {
    super(props);
    this.state = {
      currentDate: new Date()
    };
  }

  // 2. Lifecycle method: Runs once the component is mounted to the DOM
  componentDidMount() {
    this.timerID = setInterval(() => {
      // Update state using this.setState() to trigger a re-render
      this.setState({ currentDate: new Date() });
    }, 1000);
  }

  // 3. Lifecycle method: Cleans up the timer when the component is destroyed
  componentWillUnmount() {
    clearInterval(this.timerID);
  }

  render() {
    return (
      <div style={{ textAlign: "center", marginTop: "50px", fontFamily: "Arial, sans-serif" }}>
        <h2>Live System Date and Time (Class Component State)</h2>
        <p style={{ fontSize: "26px", fontWeight: "bold", color: "#2980b9" }}>
          {this.state.currentDate.toLocaleString()}
        </p>
      </div>
    );
  }
}

export default App;

Create a react application to create a two buttons and handle the click events. The click event of first display the current system date and time using event listener (Normal functions). For click event of second button display "Good Bye" by using event listener (arrow functions)
(Handling the events in React, Binding event listeners)
import React, { Component } from 'react';

class App extends Component {
  constructor(props) {
    super(props);
    this.state = {
      displayMessage: ""
    };

    // Explicitly binding the normal function to 'this'
    this.handleDateTimeClick = this.handleDateTimeClick.bind(this);
  }

  // 1. Normal function for Button 1 (requires binding in class components)
  handleDateTimeClick() {
    const currentDateTime = new Date().toLocaleString();
    this.setState({ displayMessage: `Current Date & Time: ${currentDateTime}` });
  }

  // 2. Arrow function for Button 2 (automatically binds 'this')
  handleGoodbyeClick = () => {
    this.setState({ displayMessage: "Good Bye!" });
  };

  render() {
    return (
      <div>
        <h2>Event Handling and Binding in React</h2>
        
        {/* Button 1 using a normal function with binding */}
        <button onClick={this.handleDateTimeClick}>
          Show Date and Time
        </button>
        
        {/* Button 2 using an arrow function */}
        <button onClick={this.handleGoodbyeClick}>
          Say Good Bye
        </button>

        <p>{this.state.displayMessage}</p>
      </div>
    );
  }
}

export default App;

import React, { useState } from 'react';

function App() {
  const [input, setInput] = useState('');

  // Append values to input string
  const handleClick = (value) => {
    setInput((prev) => prev + value);
  };

  // Clear display
  const handleClear = () => {
    setInput('');
  };

  // Evaluate the expression using eval()
  const handleCalculate = () => {
    try {
      // eval() computes the string mathematical expression safely here
      const result = eval(input);
      setInput(String(result));
    } catch (error) {
      setInput('Error');
    }
  };

  // React Inline Styling Objects
  const styles = {
    container: {
      width: '260px',
      margin: '40px auto',
      padding: '20px',
      border: '1px solid #ccc',
      borderRadius: '5px',
      fontFamily: 'Arial, sans-serif',
      backgroundColor: '#f9f9f9'
    },
    display: {
      width: '100%',
      height: '40px',
      marginBottom: '15px',
      fontSize: '18px',
      textAlign: 'right',
      padding: '5px',
      boxSizing: 'border-box'
    },
    buttonGrid: {
      display: 'grid',
      gridTemplateColumns: 'repeat(4, 1fr)',
      gap: '8px'
    },
    button: {
      padding: '12px',
      fontSize: '16px',
      cursor: 'pointer'
    },
    clearButton: {
      padding: '12px',
      fontSize: '16px',
      cursor: 'pointer',
      gridColumn: 'span 4',
      backgroundColor: '#f2dede',
      border: '1px solid #ebccd1'
    }
  };

  const buttons = ['7', '8', '9', '/', '4', '5', '6', '*', '1', '2', '3', '-', '0', '.', '+'];

  return (
    <div style={styles.container}>
      <h2>Calculator</h2>
      
      {/* Display screen */}
      <input type="text" value={input} readOnly style={styles.display} />

      {/* Button Grid */}
      <div style={styles.buttonGrid}>
        {buttons.map((btn, index) => (
          <button key={index} onClick={() => handleClick(btn)} style={styles.button}>
            {btn}
          </button>
        ))}
        
        {/* Equals Button */}
        <button onClick={handleCalculate} style={{ ...styles.button, backgroundColor: '#dff0d8' }}>
          =
        </button>

        {/* Clear Button */}
        <button onClick={handleClear} style={styles.clearButton}>
          Clear
        </button>
      </div>
    </div>
  );
}

import React, { useState } from 'react';

function App() {
  // 1. Initialize state for all form fields
  const [formData, setFormData] = useState({
    username: '',
    email: '',
    password: '',
    course: 'Computer Science'
  });

  const [submittedData, setSubmittedData] = useState(null);

  // 2. Handle change for inputs and select elements
  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData({
      ...formData,
      [name]: value
    });
  };

  // 3. Handle form submission
  const handleSubmit = (e) => {
    e.preventDefault(); // Prevents page reload on submit
    setSubmittedData(formData);
  };

  return (
    <div style={{ margin: '30px', fontFamily: 'Arial, sans-serif' }}>
      <h2>Student Registration Form</h2>
      
      <form onSubmit={handleSubmit}>
        <div style={{ marginBottom: '10px' }}>
          <label>Username: </label><br />
          <input
            type="text"
            name="username"
            value={formData.username}
            onChange={handleChange}
            required
          />
        </div>

        <div style={{ marginBottom: '10px' }}>
          <label>Email: </label><br />
          <input
            type="email"
            name="email"
            value={formData.email}
            onChange={handleChange}
            required
          />
        </div>

        <div style={{ marginBottom: '10px' }}>
          <label>Password: </label><br />
          <input
            type="password"
            name="password"
            value={formData.password}
            onChange={handleChange}
            required
          />
        </div>

        <div style={{ marginBottom: '15px' }}>
          <label>Course: </label><br />
          <select name="course" value={formData.course} onChange={handleChange}>
            <option value="Computer Science">Computer Science</option>
            <option value="Information Technology">Information Technology</option>
            <option value="Electronics">Electronics</option>
          </select>
        </div>

        <button type="submit">Register</button>
      </form>

      {/* Display submitted data summary */}
      {submittedData && (
        <div style={{ marginTop: '20px', padding: '15px', border: '1px solid #ccc', backgroundColor: '#f9f9f9' }}>
          <h3>Registration Successful!</h3>
          <p><strong>Username:</strong> {submittedData.username}</p>
          <p><strong>Email:</strong> {submittedData.email}</p>
          <p><strong>Course:</strong> {submittedData.course}</p>
        </div>
      )}
    </div>
  );
}
